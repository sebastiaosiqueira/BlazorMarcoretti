    private MenuPermissaoDTO _perm;

    protected override async Task OnInitializedAsync()
    {
        // Pega as permissões já cacheadas no serviço para esta página específica
        var todas = await PermissionService.GetUserPermissionsAsync(meuPerfilId);
        // Busca a permissão deste menu específico (pelo URL ou ID)
        _perm = todas.SelectMany(x => x.SubMenus.Append(x))
                     .FirstOrDefault(m => m.Url == "/laboratorios");
    }
}
